## المعادلات في مكان واحد

```ts
// الدونت في Filee (نطة 8.5%)
const easeOutBack = (t: number, c1 = 1.55) => 1 + (c1 + 1) * (t - 1) ** 3 + c1 * (t - 1) ** 2;

// BackEaseOut الجاهز في Avalonia (نطة 38%، للمسافات الصغيرة)
const avaloniaBackOut = (t: number) => { const p = 1 - t; return 1 - p * (p * p - Math.sin(p * Math.PI)); };

const easeInCubic = (t: number) => t * t * t;
const cubicEaseOut = (t: number) => 1 - (1 - t) ** 3;

// spring: خطوة واحدة صغيرة
velocity += (stiffness * (target - value) - damping * velocity) * h;
value += velocity * h;
```
