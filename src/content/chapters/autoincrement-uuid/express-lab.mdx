## الإنشاء والقراءة

<File name="src/server.js">

```js
import express from "express";
import { pool } from "./db.js";
import { uuidv7 } from "./uuidv7.js";

const app = express();
app.use(express.json());

const UUID_RE =
  /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;

app.post("/users", async (req, res, next) => {
  try {
    const name = String(req.body?.name ?? "").trim();
    if (!name) return res.status(400).json({ error: "name_required" });

    const publicId = uuidv7();
    const { rows } = await pool.query(
      `INSERT INTO users (public_id, name)
       VALUES ($1, $2)
       RETURNING public_id`,
      [publicId, name],
    );
    res.status(201).json({ id: rows[0].public_id, name });
  } catch (err) {
    if (err.code === "23505") return res.status(409).json({ error: "conflict" });
    next(err);
  }
});

app.get("/users/:publicId", async (req, res, next) => {
  try {
    const publicId = req.params.publicId;
    if (!UUID_RE.test(publicId)) return res.status(400).json({ error: "bad_id" });

    const { rows } = await pool.query(
      "SELECT public_id, name FROM users WHERE public_id = $1",
      [publicId],
    );
    const user = rows[0];
    if (!user) return res.status(404).json({ error: "not_found" });

    // لو المورد خاص: افحص الجلسة هنا. صعوبة الـ UUID مش الفحص.
    res.json({ id: user.public_id, name: user.name });
  } catch (err) {
    next(err);
  }
});

app.use((err, _req, res, _next) => {
  console.error(err);
  res.status(500).json({ error: "internal" });
});

app.listen(3000, () => console.log("users api on :3000"));
```

</File>

`RETURNING` بيرجع المعرّف العام بس. الـ `id` الداخلي موجود في الصف ومش في الرد. `23505` تصادم الـ unique، نادر، والرد ٤٠٩ أحسن من ٥٠٠. فحص شكل الـ UUID قبل الكويري عشان نص عشوائي مايبقاش خطأ من الدرايفر.

<Box type="warn" title="المعمل مش نظام دخول">

مفيش جلسة في الكود ده. البروفايل هنا عام. لو الصفحة خاصة، حط الفحص بعد ما تجيب الصف، ومتاخدش هوية اليوزر من هيدر هو اللي كاتبه. ومتبعتش `id` الـ bigint في JSON "عشان الفرونت محتاجه في اللوج".

</Box>

<Box type="tip" title="لو عايز Postgres يولّد v7">

في PostgreSQL 18 تقدر تخلي الافتراضي `uuidv7()` وتسيب التطبيق من غير مولّد. لسه `INSERT ... RETURNING` عشان تعرف الرقم. قبل ١٨، الدالة دي مش مضمونة. راجع الإصدار.

</Box>

## 🧪 Lab

شغّل الجدول، وبعدين:

```bash
curl -s localhost:3000/users -H 'content-type: application/json' -d '{"name":"منى"}'
curl -s localhost:3000/users/PASTE-THE-PUBLIC-ID
curl -s -o /dev/null -w '%{http_code}\n' localhost:3000/users/123456789
```

الأول يرد `id` طويل مش رقم صغير. التاني نفس المعرّف. التالت ٤٠٠ لأن الشكل مش UUID، مش ٥٠٠ من Postgres. في الداتابيز، `SELECT id, public_id FROM users` يوريك الرقمين، والـ API مش شايف غير واحد.
