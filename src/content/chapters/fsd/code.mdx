## فيتشر الإضافة

<File name="features/add-to-cart/model/use-add-to-cart.ts">

```ts
import { useMutation, useQueryClient } from '@tanstack/react-query';
import { addCartItem } from '@/entities/cart';

export function useAddToCart() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: addCartItem,

    onSuccess: async () => {
      await queryClient.invalidateQueries({
        queryKey: ['cart', 'current'],
      });
    },
  });
}
```

</File>

<File name="features/add-to-cart/ui/add-to-cart-button.tsx">

```tsx
import { useAddToCart } from '../model/use-add-to-cart';

interface Props {
  productId: string;
  disabled?: boolean;
}

export function AddToCartButton({
  productId,
  disabled,
}: Props) {
  const mutation = useAddToCart();

  return (
    <button
      disabled={disabled || mutation.isPending}
      onClick={() =>
        mutation.mutate({
          productId,
          quantity: 1,
        })
      }
    >
      {mutation.isPending
        ? 'Adding...'
        : 'Add to Cart'}
    </button>
  );
}
```

</File>
